import json
import os
import random

class Flashcard:
    def __init__(self, front, back):
        self.front = front
        self.back = back

class Deck:
    def __init__(self, name):
        self.name = name
        self.flashcards = []

    def add_flashcard(self, flashcard):
        self.flashcards.append(flashcard)

    def save(self, filename):
        with open(filename, 'w') as file:
            json.dump([{'front': card.front, 'back': card.back} for card in self.flashcards], file)

    @staticmethod
    def load(filename):
        with open(filename, 'r') as file:
            data = json.load(file)
            deck = Deck(filename.split('.')[0])
            for item in data:
                deck.add_flashcard(Flashcard(item['front'], item['back']))
            return deck

    def review(self):
        if not self.flashcards:
            print("No flashcards to review.")
            return
        random.shuffle(self.flashcards)
        for card in self.flashcards:
            input(f"Front: {card.front}\nPress Enter to see the back...")
            print(f"Back: {card.back}\n")

def main():
    decks = {}
    while True:
        print("\nCLI Flashcard Study Tool")
        print("1. Create a new deck")
        print("2. Add a flashcard to a deck")
        print("3. Review a deck")
        print("4. Load a deck")
        print("5. Exit")
        choice = input("Choose an option: ")

        if choice == '1':
            name = input("Enter deck name: ")
            if name in decks:
                print("Deck already exists.")
            else:
                decks[name] = Deck(name)
                print(f"Deck '{name}' created.")

        elif choice == '2':
            name = input("Enter deck name: ")
            if name not in decks:
                print("Deck does not exist.")
            else:
                front = input("Enter front of flashcard: ")
                back = input("Enter back of flashcard: ")
                decks[name].add_flashcard(Flashcard(front, back))
                print("Flashcard added.")

        elif choice == '3':
            name = input("Enter deck name: ")
            if name not in decks:
                print("Deck does not exist.")
            else:
                decks[name].review()

        elif choice == '4':
            filename = input("Enter filename to load: ")
            if not os.path.exists(filename):
                print("File does not exist.")
            else:
                deck = Deck.load(filename)
                decks[deck.name] = deck
                print(f"Deck '{deck.name}' loaded.")

        elif choice == '5':
            for name, deck in decks.items():
                deck.save(f"{name}.json")
            print("Exiting...")
            break

        else:
            print("Invalid choice. Please try again.")

if __name__ == "__main__":
    main()
