import argparse

def load_dictionary(file_path):
    dictionary = {}
    with open(file_path, 'r') as file:
        for line in file:
            word, definition = line.strip().split(':', 1)
            dictionary[word.strip()] = definition.strip()
    return dictionary

def lookup_definition(dictionary, word):
    return dictionary.get(word, f"Definition for '{word}' not found.")

def main():
    parser = argparse.ArgumentParser(description="CLI Dictionary Lookup Tool")
    parser.add_argument('word', type=str, help='The word to look up')
    parser.add_argument('--file', type=str, default='dictionary.txt', help='Path to the dictionary file')
    args = parser.parse_args()

    dictionary = load_dictionary(args.file)
    definition = lookup_definition(dictionary, args.word)
    print(definition)

if __name__ == "__main__":
    main()
