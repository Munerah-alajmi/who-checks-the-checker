import sys

def convert_weight(value, source_unit, target_unit):
    conversion_factors = {
        'kg_to_lb': 2.20462,
        'g_to_oz': 0.035274,
        'lb_to_kg': 1 / 2.20462,
        'oz_to_g': 1 / 0.035274,
        'kg_to_g': 1000,
        'g_to_kg': 1 / 1000,
        'lb_to_oz': 16,
        'oz_to_lb': 1 / 16
    }
    
    key = f"{source_unit}_to_{target_unit}"
    if key not in conversion_factors:
        return "Invalid units provided."
    
    return value * conversion_factors[key]

def main():
    if len(sys.argv) != 4:
        print("Usage: python unit_converter.py <value> <source_unit> <target_unit>")
        sys.exit(1)
    
    try:
        value = float(sys.argv[1])
    except ValueError:
        print("Please provide a valid number for the value.")
        sys.exit(1)
    
    source_unit = sys.argv[2].lower()
    target_unit = sys.argv[3].lower()
    
    result = convert_weight(value, source_unit, target_unit)
    print(f"{value} {source_unit} is {result} {target_unit}")

if __name__ == "__main__":
    main()
